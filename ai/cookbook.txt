4. REACT TO CHANGES IN LIVEVIEW
-------------------------------------------------------------------------------
    <.web_multiselect id="tags" hook={true} options={@opts} value={@selected} />

    def handle_event("web_multiselect:change", %{"id" => "tags", "values" => v}, socket) do
      {:noreply, assign(socket, :selected, v)}
    end

5. SERVER-AUTHORITATIVE RULE — "max 3" (allow optimistically, then correct)
-------------------------------------------------------------------------------
    <.web_multiselect id="tags" hook={true} multiple={true} options={@opts} />

    def handle_event("web_multiselect:change", %{"id" => "tags", "values" => v}, socket)
        when length(v) > 3 do
      kept = Enum.take(v, 3)
      {:noreply,
       socket
       |> assign(:tags, kept)
       |> put_flash(:error, "Max 3 — reverted.")
       |> Keenmate.WebMultiselect.push_update("tags", value: kept)}
    end

    def handle_event("web_multiselect:change", %{"id" => "tags", "values" => v}, socket) do
      {:noreply, assign(socket, :tags, v)}
    end

6. CASCADING SELECTS — countries drive a grouped city list
   (country is multi-select; city = union of all selected countries' cities,
    grouped by country; existing valid picks preserved)
-------------------------------------------------------------------------------
    @cities %{
      "cz" => ["Prague", "Brno"], "sk" => ["Bratislava"],
      "pl" => ["Warsaw", "Kraków"], "de" => ["Berlin", "Munich"]
    }

    <.web_multiselect id="country" hook={true} multiple={true} options={@countries} />
    <.web_multiselect id="city" hook={true} multiple={true} allow_groups={true}
      placeholder="Cities (updated from server)" />

    def handle_event("web_multiselect:change", %{"id" => "country", "values" => codes}, socket) do
      options =
        for code <- codes, city <- Map.get(@cities, code, []),
          do: %{value: city, label: city, group: country_label(code)}

      valid = MapSet.new(options, & &1.value)
      kept  = Enum.filter(socket.assigns.city, &MapSet.member?(valid, &1))

      {:noreply,
       socket
       |> assign(:country, codes)
       |> assign(:city, kept)
       |> Keenmate.WebMultiselect.push_update("city", options: options, value: kept)}
    end

    def handle_event("web_multiselect:change", %{"id" => "city", "values" => v}, socket) do
      {:noreply, assign(socket, :city, v)}
    end

7. ASYNC SERVER-SIDE SEARCH (zero JS)
-------------------------------------------------------------------------------
    <.web_multiselect id="repos" hook={true}
      search_event="search_repos" search_debounce={250}
      min_search_length={2} search_placeholder="Search GitHub..." />

    def handle_event("search_repos", %{"id" => "repos", "query" => q}, socket) do
      results =
        q |> MyApp.GitHub.search_repos() |> Enum.map(&%{value: &1.id, label: &1.full_name})
      {:reply, %{results: results}, socket}   # MUST be {:reply, ...}
    end

8. RESET / CLEAR FROM ANOTHER BUTTON
-------------------------------------------------------------------------------
    <button phx-click="reset">Clear tags</button>

    def handle_event("reset", _p, socket) do
      {:noreply,
       socket
       |> assign(:tags, [])
       |> Keenmate.WebMultiselect.push_update("tags", value: [])}
    end

-------------------------------------------------------------------------------
9. LOAD OPTIONS AFTER MOUNT (lazy / from DB)
-------------------------------------------------------------------------------
    def mount(_p, _s, socket) do
      if connected?(socket), do: send(self(), :load_tags)
      {:ok, assign(socket, tags: [])}
    end

    def handle_info(:load_tags, socket) do
      {:noreply, Keenmate.WebMultiselect.push_update(socket, "tags", options: MyApp.tags())}
    end
